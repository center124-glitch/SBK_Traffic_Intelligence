// ---------------------------------------------------------------- entrypoint
function doGet(e) {
  var p  = (e && e.parameter) ? e.parameter : {};
  var cb = p.callback || null;
  try {
    var mode    = p.mode || 'summary';
    var nocache = p.nocache === '1';
    var ids     = p.folderIds ? String(p.folderIds).split('|') : DEFAULT_FOLDER_IDS.slice();
    ids = ids.map(trim_).filter(nonEmpty_);

    /* ทางลัดสำหรับ mode=summary ที่ระบุ ids มาแล้ว
       ไม่ต้องสแกนทั้ง 60 ไฟล์เพื่อหาแค่ 2 ไฟล์ — เปิดจาก id ตรง ๆ เร็วกว่ามาก
       แต่ต้องตรวจก่อนว่าไฟล์นั้นอยู่ใต้โฟลเดอร์ที่อนุญาตจริง เพราะ Web App นี้
       ตั้ง "Anyone" ถ้าไม่ตรวจ ใครก็ส่ง id ไฟล์อะไรก็ได้มาอ่านผ่านสิทธิ์ของเจ้าของสคริปต์ */
    if (mode === 'summary' && p.ids) {
      var wantFast = String(p.ids).split('|').map(trim_).filter(nonEmpty_);
      if (wantFast.length && wantFast.length <= MAX_SUMMARY_FILES) {
        var allowed = {};
        for (var ai = 0; ai < ids.length; ai++) allowed[ids[ai]] = true;
        var fastOut = [], fastFail = [];
        for (var fi = 0; fi < wantFast.length; fi++) {
          try {
            var file = DriveApp.getFileById(wantFast[fi]);
            if (!isUnderAllowedFolder_(file, allowed)) {
              fastFail.push({ id: wantFast[fi], error: 'ไฟล์นี้ไม่ได้อยู่ในโฟลเดอร์ที่อนุญาต' });
              continue;
            }
            fastOut.push(summarizeFile_({
              id: file.getId(), name: file.getName(), folder: '',
              updated: file.getLastUpdated().getTime(), size: file.getSize(),
              isSheet: file.getMimeType() === MimeType.GOOGLE_SHEETS
            }, nocache));
          } catch (ferr) {
            fastFail.push({ id: wantFast[fi], error: String(ferr) });
          }
        }
        return respond_({ ok: true, version: SCRIPT_VERSION, mode: 'summary',
          runAs: effectiveUser_(), folders: [], skipped: [], failed: fastFail, files: fastOut }, cb);
      }
    }

    var scan = scanFolders_(ids);

    if (mode === 'diag') {
      return respond_({
        ok: true, version: SCRIPT_VERSION, mode: 'diag',
        runAs: effectiveUser_(), folders: scan.folders,
        skipped: scan.skipped, fileCount: scan.files.length,
        files: scan.files.map(function (f) { return { name: f.name, folder: f.folder }; })
      }, cb);
    }

    if (mode === 'list') {
      /* อ่านแค่ metadata — ไม่เปิดสเปรดชีตเลยแม้แต่ไฟล์เดียว จึงเสร็จในไม่กี่วินาที
         แม้โฟลเดอร์จะมี 60 ไฟล์ และแนบเดือน/ปีมาให้ ฝั่งเว็บจะได้สร้างเมนูเลือกช่วงได้ทันที */
      return respond_({
        ok: true, version: SCRIPT_VERSION, mode: 'list',
        runAs: effectiveUser_(), folders: scan.folders, skipped: scan.skipped,
        files: scan.files.map(function (f) {
          var per = parsePeriod_(f.name);
          return { id: f.id, name: f.name, folder: f.folder, updated: f.updated, size: f.size,
                   month: per.month, year: per.year, label: per.label };
        })
      }, cb);
    }

    // ---- mode = summary ----------------------------------------------------
    var want = p.ids ? String(p.ids).split('|').map(trim_).filter(nonEmpty_) : null;

    /* บังคับให้ระบุ ids เสมอ
       ถ้าปล่อยให้สรุปทั้งโฟลเดอร์ได้ คำขอจะใช้เวลาเกินลิมิต 6 นาทีแล้วตายเงียบ ๆ
       ซึ่งฝั่งเว็บแยกไม่ออกจาก "เชื่อมต่อไม่ได้" — เป็นที่มาของปัญหาเดิม */
    if (!want || !want.length) {
      return respond_({ ok: false, version: SCRIPT_VERSION, runAs: effectiveUser_(),
        error: 'ต้องระบุพารามิเตอร์ ids (เรียก mode=list ก่อนเพื่อดูรายชื่อไฟล์) ' +
               'โฟลเดอร์นี้มี ' + scan.files.length + ' ไฟล์ การสรุปทั้งหมดพร้อมกันจะเกินเวลาทำงานของ Apps Script' }, cb);
    }
    if (want.length > MAX_SUMMARY_FILES) {
      return respond_({ ok: false, version: SCRIPT_VERSION, runAs: effectiveUser_(),
        error: 'ขอข้อมูลครั้งละไม่เกิน ' + MAX_SUMMARY_FILES + ' ไฟล์ (ขอมา ' + want.length + ')' }, cb);
    }

    var byId = {};
    for (var k = 0; k < scan.files.length; k++) byId[scan.files[k].id] = scan.files[k];

    var out = [], failed = [];
    for (var i = 0; i < want.length; i++) {
      var meta = byId[want[i]];
      if (!meta) { failed.push({ id: want[i], error: 'ไม่พบไฟล์นี้ในโฟลเดอร์ที่สแกน' }); continue; }
      try {
        out.push(summarizeFile_(meta, nocache));
      } catch (err) {
        failed.push({ name: meta.name, error: String(err) });
      }
    }
    return respond_({
      ok: true, version: SCRIPT_VERSION, mode: 'summary',
      runAs: effectiveUser_(), folders: scan.folders,
      skipped: scan.skipped, failed: failed, files: out
    }, cb);

  } catch (err) {
    return respond_({ ok: false, version: SCRIPT_VERSION, error: String(err),
                      runAs: effectiveUser_() }, cb);
  }
}

// ------------------------------------------------------------ folder scanning
function scanFolders_(ids) {
  var files = [], skipped = [], folders = [];
  for (var i = 0; i < ids.length; i++) {
    var fid = ids[i];
    try {
      var folder = DriveApp.getFolderById(fid);
      var n0 = files.length, s0 = skipped.length;
      collect_(folder, 0, files, skipped);
      folders.push({ id: fid, name: folder.getName(),
                     matched: files.length - n0, skipped: skipped.length - s0 });
    } catch (err) {
      folders.push({ id: fid, name: null, matched: 0, skipped: 0, error: String(err) });
    }
  }
  return { files: dedupe_(files), skipped: skipped, folders: folders };
}

function collect_(folder, depth, files, skipped) {
  var it = folder.getFiles();
  while (it.hasNext()) {
    var f = it.next(), nm = f.getName();
    if (!isSpreadsheet_(f))      { skipped.push({ name: nm, folder: folder.getName(), reason: 'ไม่ใช่สเปรดชีต' }); continue; }
    if (!isTrafficFile_(nm))     { skipped.push({ name: nm, folder: folder.getName(), reason: 'ชื่อไฟล์ไม่เข้าเกณฑ์' }); continue; }
    files.push({ id: f.getId(), name: nm, folder: folder.getName(),
                 updated: f.getLastUpdated().getTime(), size: f.getSize(),
                 isSheet: f.getMimeType() === MimeType.GOOGLE_SHEETS });
  }
  if (depth >= MAX_DEPTH) return;
  var sub = folder.getFolders();
  while (sub.hasNext()) {
    var sf = sub.next();
    // โฟลเดอร์ขึ้นต้นด้วย _ เป็นที่เก็บแคชของระบบอื่น ไม่ใช่ข้อมูลต้นทาง — ข้ามไปเลย
    if (SKIP_FOLDER_RE.test(sf.getName())) continue;
    collect_(sf, depth + 1, files, skipped);
  }
}

/** ไฟล์นี้อยู่ใต้โฟลเดอร์ที่อนุญาตหรือไม่ (ไล่ขึ้นไปหาแม่ไม่เกิน MAX_DEPTH ชั้น) */
function isUnderAllowedFolder_(file, allowed) {
  var level = [file], depth = 0;
  while (level.length && depth <= MAX_DEPTH + 1) {
    var next = [];
    for (var i = 0; i < level.length; i++) {
      var it = level[i].getParents();
      while (it.hasNext()) {
        var par = it.next();
        if (allowed[par.getId()]) return true;
        next.push(par);
      }
    }
    level = next; depth++;
  }
  return false;
}

function isSpreadsheet_(file) {
  var m = file.getMimeType();
  return m === MimeType.GOOGLE_SHEETS || m === MimeType.MICROSOFT_EXCEL ||
         /\.(xlsx|xls)$/i.test(file.getName());
}

/** ชื่อไฟล์ต้องสื่อถึงรายงานยอดลูกค้า+รถ และระบุเดือน/ปีได้ */
function isTrafficFile_(name) {
  var n = String(name).replace(/\s+/g, ' ');
  var topic = /ลูกค้า/.test(n) && /รถ/.test(n);
  var month = /(^|\s)(ม\.?ค|ก\.?พ|มี\.?ค|เม\.?ย|พ\.?ค|มิ\.?ย|ก\.?ค|ส\.?ค|ก\.?ย|ต\.?ค|พ\.?ย|ธ\.?ค)\.?(\s|$|\d)/.test(n) ||
              /(มกราคม|กุมภาพันธ์|มีนาคม|เมษายน|พฤษภาคม|มิถุนายน|กรกฎาคม|สิงหาคม|กันยายน|ตุลาคม|พฤศจิกายน|ธันวาคม)/.test(n);
  var year  = /\b(25\d{2}|\d{2})\b/.test(n);
  return topic && year && (month || /เดือน/.test(n));
}

function dedupe_(files) {
  var by = {}, out = [];
  for (var i = 0; i < files.length; i++) {
    var f = files[i];
    var k = String(f.name).replace(/\.(xlsx|xls)$/i, '').replace(/\s+/g, ' ').trim();
    if (!by[k] || f.updated > by[k].updated) by[k] = f;
  }
  for (var k2 in by) if (by.hasOwnProperty(k2)) out.push(by[k2]);
  out.sort(function (a, b) { return a.name < b.name ? -1 : 1; });
  return out;
}

/**
 * สรุป 1 ไฟล์ (1 เดือน) → อ็อบเจกต์เล็ก ๆ
 * แคชด้วยคีย์ที่ผูกกับเวลาแก้ไขไฟล์ ถ้าไฟล์ถูกแก้ แคชจะไม่ชนกันเอง
 */
function summarizeFile_(meta, nocache) {
  var cache = CacheService.getScriptCache();
  var key   = 'sbk50:' + meta.id + ':' + meta.updated;
  if (!nocache) {
    var hit = getChunked_(cache, key);
    if (hit) { var o = JSON.parse(hit); o.cached = true; return o; }
  }

  var conv = null, ss;
  if (meta.isSheet) {
    ss = SpreadsheetApp.openById(meta.id);
  } else {
    conv = openXlsxAsSheet_(meta);     // คืน {ss, tmpId} — ต้องลบ tmpId ทิ้งเสมอ
    ss = conv.ss;
  }
  var sheets = ss.getSheets();
  var period = parsePeriod_(meta.name);

  var raw = [], sheetErrors = [], notes = [];
  /* รวมรายชื่อประตูจาก "ทุกชีต" ไม่ใช่ชีตแรกชีตเดียว
     ตลอด 5 ปีมีประตูเปิดใหม่/ปิดไป ถ้ายึดชีตแรกแล้วรวมยอดตาม "ลำดับแถว"
     ยอดของประตู A ในเดือนหนึ่งจะไปบวกกับประตู B ในอีกเดือนหนึ่งโดยไม่มีใครรู้ */
  var gateIdx = {}, gates = [];
  function gateSlot_(name) {
    if (gateIdx[name] === undefined) { gateIdx[name] = gates.length; gates.push(name); }
    return gateIdx[name];
  }

  for (var i = 0; i < sheets.length; i++) {
    try {
      var r = summarizeSheet_(sheets[i], period);
      if (r) raw.push(r);
    } catch (err) {
      sheetErrors.push({ sheet: sheets[i].getName(), error: String(err) });
    }
  }

  // ลงทะเบียนชื่อประตูทั้งหมดก่อน แล้วค่อยจัดยอดให้ตรงช่อง
  for (var g1 = 0; g1 < raw.length; g1++) {
    var L = raw[g1].labelsIn || [];
    for (var g2 = 0; g2 < L.length; g2++) gateSlot_(L[g2]);
    var L2 = raw[g1].labelsOut || [];
    for (var g3 = 0; g3 < L2.length; g3++) gateSlot_(L2[g3]);
  }

  var days = [], seen = {}, emptyDays = [];
  for (var k = 0; k < raw.length; k++) {
    var d = raw[k].day;

    /* วันที่ไม่มีข้อมูลเลย (ชีตเปล่าที่เตรียมไว้ล่วงหน้า) ต้องไม่ถูกนับเป็น 1 วัน
       ไม่งั้นตัวหารของค่าเฉลี่ยจะเกินจริง แล้วค่าเฉลี่ยต่อวันจะต่ำกว่าความเป็นจริง */
    if (!d.cIn && !d.cOut && !d.vIn24 && !d.vOut24 && !d.mc && !d.dlv) {
      emptyDays.push(d.d);
      continue;
    }
    // วันซ้ำ (ชีตชื่อซ้ำ) — เก็บอันหลังสุด แล้วแจ้งไว้
    if (seen[d.d] !== undefined) {
      notes.push('วันที่ ' + d.d + ' มีมากกว่า 1 ชีต — ใช้ชีตล่าสุด');
      days[seen[d.d]] = alignGates_(d, raw[k], gateIdx, gates.length);
      continue;
    }
    seen[d.d] = days.length;
    days.push(alignGates_(d, raw[k], gateIdx, gates.length));
  }
  if (emptyDays.length) notes.push('ข้ามวันที่ไม่มีข้อมูล ' + emptyDays.length + ' วัน (' + emptyDays.join(',') + ')');
  days.sort(function (a, b) { return a.d - b.d; });

  var res = {
    id: meta.id, name: meta.name, folder: meta.folder, updated: meta.updated,
    month: period.month, year: period.year, label: period.label,
    gates: gates, days: days, sheetErrors: sheetErrors, notes: notes, cached: false
  };
  try { putChunked_(cache, key, JSON.stringify(res)); } catch (e) {}

  /* ลบไฟล์ชั่วคราวที่เกิดจากการแปลง .xlsx ทุกครั้ง
     เวอร์ชันก่อนสร้างแล้วไม่ลบ — เปิดแอปวันละไม่กี่ครั้งก็ถมไดรฟ์เต็มได้ */
  if (conv && conv.tmpId) {
    try { DriveApp.getFileById(conv.tmpId).setTrashed(true); } catch (e) {}
  }
  return res;
}

/**
 * จัดยอดรายประตูของ 1 วัน ให้ตรงช่องตาม "ชื่อประตู" ในสารบบกลางของไฟล์
 * ประตูที่วันนั้นไม่มี จะได้ 0 — ฝั่งเว็บจึงบวกตามลำดับช่องได้อย่างปลอดภัย
 */
function alignGates_(day, r, gateIdx, n) {
  var gi = [], go = [], i;
  for (i = 0; i < n; i++) { gi.push(0); go.push(0); }
  var Li = r.labelsIn || [], Vi = day.gIn || [];
  for (i = 0; i < Li.length; i++) { var p1 = gateIdx[Li[i]]; if (p1 !== undefined) gi[p1] += (Vi[i] || 0); }
  var Lo = r.labelsOut || [], Vo = day.gOut || [];
  for (i = 0; i < Lo.length; i++) { var p2 = gateIdx[Lo[i]]; if (p2 !== undefined) go[p2] += (Vo[i] || 0); }
  day.gIn = gi; day.gOut = go;
  return day;
}

/**
 * ไฟล์ .xlsx แท้: ต้องแปลงเป็น Google Sheets ชั่วคราวก่อนจึงอ่านค่าได้
 * ต้องเปิด "Drive API" ใน Services ของโปรเจกต์ Apps Script ก่อน ไม่งั้น Drive จะ undefined
 * — ข้อความ error ปกติจะเป็น "Drive is not defined" ซึ่งเดาสาเหตุไม่ออก จึงดักไว้ให้ชัด
 */
function openXlsxAsSheet_(meta) {
  if (typeof Drive === 'undefined' || !Drive.Files) {
    throw new Error('ไฟล์ "' + meta.name + '" เป็น .xlsx ซึ่งต้องเปิดบริการ Drive API ก่อน ' +
      '(ในตัวแก้ไข Apps Script: Services → + → Drive API → Add) ' +
      'หรือเปลี่ยนไฟล์นี้เป็น Google Sheets แทน');
  }
  var blob = DriveApp.getFileById(meta.id).getBlob();
  var tmp  = Drive.Files.insert(
    { title: '__sbk_tmp_' + meta.id + '_' + Date.now(), mimeType: MimeType.GOOGLE_SHEETS },
    blob, { convert: true });
  return { ss: SpreadsheetApp.openById(tmp.id), tmpId: tmp.id };
}

/** สรุป 1 ชีต = 1 วัน  (คืน null ถ้าชีตนี้ไม่ใช่ตารางข้อมูลรายวัน) */
function summarizeSheet_(sh, period) {
  var v = sh.getDataRange().getValues();       // อ่านทีเดียวจบ — เร็วกว่าอ่านทีละเซลล์มาก
  if (!v || v.length < 40) return null;

  var idx = indexLabels_(v);
  if (idx[LBL.custIn] == null) return null;    // ไม่มีบล็อกสรุป = ไม่ใช่ชีตข้อมูล

  var dayNo = parseDayNo_(sh.getName(), v);
  if (!dayNo) return null;

  /* --- หาหัวตารางรายชั่วโมงทั้ง 4 ตารางในชีต ---------------------------
     ห้ามค้นด้วยข้อความ "10.00 น." เฉย ๆ เพราะตารางรถยนต์ (01.00–24.00 น.)
     ก็มีคอลัมน์ 10.00 น. อยู่กลางตารางด้วย จะจับผิดตารางทันที
     จึงยึดจากแถวหัวตาราง (คอลัมน์ B = "จุดประตู") แล้วนับจำนวนช่องชั่วโมง:
     15 ช่อง = ตารางลูกค้า, 24 ช่อง = ตารางยานพาหนะ
     ลำดับในไฟล์คือ ลูกค้าเข้า → รถเข้า → ลูกค้าออก → รถออก */
  var heads = findTableHeaders_(v);
  var custHeads = [], vehHeads = [];
  for (var hi = 0; hi < heads.length; hi++) {
    (heads[hi].nHours >= 20 ? vehHeads : custHeads).push(heads[hi]);
  }

  var custInBlk  = custHeads.length > 0 ? readHourBlock_(v, custHeads[0], 15) : emptyBlock_(15);
  var vehInBlk   = vehHeads.length  > 0 ? readHourBlock_(v, vehHeads[0],  24) : emptyBlock_(24);
  var custOutBlk = custHeads.length > 1 ? readHourBlock_(v, custHeads[1], 15) : emptyBlock_(15);
  var vehOutBlk  = vehHeads.length  > 1 ? readHourBlock_(v, vehHeads[1],  24) : emptyBlock_(24);

  var lv = function (label) { var r = idx[label]; return r == null ? 0 : rowValue_(v, r); };

  var day = {
    d: dayNo,
    cIn  : lv(LBL.custIn),
    cOut : lv(LBL.custOut),
    vIn24: lv(LBL.carIn24),
    vIn  : lv(LBL.carIn1022),
    vOut24: lv(LBL.carOut24),
    vOut : lv(LBL.carOut1022),
    mc   : lv(LBL.mc),
    taxi : lv(LBL.taxi),
    dlv  : lv(LBL.dlv),
    cInH : custInBlk.hours,  cOutH: custOutBlk.hours,
    vInH : vehInBlk.hours,   vOutH: vehOutBlk.hours,
    gIn  : custInBlk.rows,   gOut : custOutBlk.rows
  };

  // ถ้าบล็อกสรุปว่าง ให้ถอยไปใช้ผลรวมรายชั่วโมงแทน (กันข้อมูลหายเงียบ ๆ)
  if (!day.cIn)   day.cIn   = sum_(day.cInH);
  if (!day.cOut)  day.cOut  = sum_(day.cOutH);
  if (!day.vIn24) day.vIn24 = sum_(day.vInH);
  if (!day.vOut24) day.vOut24 = sum_(day.vOutH);

  return { day: day, labelsIn: custInBlk.labels, labelsOut: custOutBlk.labels };
}

/** ทำดัชนี: ป้ายข้อความในคอลัมน์ B → เลขแถว (0-based) */
function indexLabels_(v) {
  var idx = {};
  for (var r = 0; r < v.length; r++) {
    var c = v[r][1];
    if (c == null) continue;
    var s = String(c).replace(/\s+/g, ' ').trim();
    for (var k in LBL) {
      if (!LBL.hasOwnProperty(k)) continue;
      var want = LBL[k].replace(/\s+/g, ' ');
      if (idx[LBL[k]] == null && s.indexOf(want) === 0) idx[LBL[k]] = r;
    }
  }
  return idx;
}

/** ค่าตัวเลขตัวแรกที่เจอในแถวสรุป (ไฟล์จริงวางไว้คอลัมน์ D แต่ไม่ยึดตายตัว) */
function rowValue_(v, r) {
  for (var c = 2; c < v[r].length; c++) {
    var n = toNum_(v[r][c]);
    if (n !== null) return n;
  }
  return 0;
}

/**
 * หาแถวหัวตารางทั้งหมดในชีต (แถวที่คอลัมน์ B เขียนว่า "จุดประตู")
 * แล้วนับว่าหัวตารางนั้นมีคอลัมน์ชั่วโมงกี่ช่อง เพื่อแยกว่าเป็นตารางลูกค้าหรือยานพาหนะ
 * คืน [{row, col, nHours}] เรียงตามลำดับที่ปรากฏในไฟล์
 */
function findTableHeaders_(v) {
  var heads = [];
  for (var r = 0; r < v.length; r++) {
    var b = v[r][1];
    if (b == null) continue;
    if (String(b).replace(/\s+/g, ' ').trim().indexOf('จุดประตู') !== 0) continue;

    // ไล่นับคอลัมน์ที่เป็นเวลา (เช่น "10.00 น.") ถัดจากคอลัมน์ B ไปทางขวา
    var start = -1, n = 0;
    for (var c = 2; c < v[r].length; c++) {
      var s = String(v[r][c] == null ? '' : v[r][c]).replace(/\s+/g, '');
      if (/^\d{1,2}\.\d{2}น\.?$/.test(s)) { if (start < 0) start = c; n++; }
      else if (start >= 0) break;                 // พ้นช่วงเวลาแล้ว (ปกติเจอ "รวม")
    }
    if (start >= 0 && n > 0) heads.push({ row: r, col: start, nHours: n });
  }
  return heads;
}

/**
 * อ่านบล็อกตารางรายชั่วโมงใต้หัวตาราง
 * คืน: hours[] ผลรวมรายชั่วโมง, rows[] ผลรวมรายแถว(ประตู/เลน), labels[] ชื่อแถว
 * หยุดเมื่อเจอแถว "รวม" หรือแถวว่าง
 */
function readHourBlock_(v, hdr, nHours) {
  var hours = zeros_(nHours), rows = [], labels = [];
  for (var r = hdr.row + 1; r < v.length; r++) {
    var nameCell = v[r][1];
    var nm = nameCell == null ? '' : String(nameCell).replace(/\s+/g, ' ').trim();
    if (nm === LBL.sumRow) break;                 // ถึงแถว "รวม" = จบบล็อก
    if (nm === '' && allBlank_(v[r], hdr.col, nHours)) break;
    if (nm === '' ) continue;
    if (nm.indexOf(LBL.delivery) === 0) continue; // แถวเดลิเวอรี่นับแยก ไม่รวมกับประตู

    var tot = 0;
    for (var h = 0; h < nHours; h++) {
      var n = toNum_(v[r][hdr.col + h]);
      if (n !== null) { hours[h] += n; tot += n; }
    }
    rows.push(tot); labels.push(nm);
  }
  return { hours: hours, rows: rows, labels: labels };
}

function emptyBlock_(n) { return { hours: zeros_(n), rows: [], labels: [] }; }

/** เลขวันจากชื่อชีต เช่น "1  พ.ค.  68" → 1 ; สำรองด้วยหัวเรื่อง B1 */
function parseDayNo_(sheetName, v) {
  var m = String(sheetName).trim().match(/^(\d{1,2})/);
  if (m) return parseInt(m[1], 10);
  var title = String(v[0] && v[0][1] ? v[0][1] : '');
  var m2 = title.match(/วันที่\s*(\d{1,2})/);
  return m2 ? parseInt(m2[1], 10) : 0;
}

/** เดือน/ปี พ.ศ. จากชื่อไฟล์ */
function parsePeriod_(name) {
  var n = String(name).replace(/\.(xlsx|xls)$/i, '').replace(/\s+/g, ' ');
  var abbr = ['ม.ค','ก.พ','มี.ค','เม.ย','พ.ค','มิ.ย','ก.ค','ส.ค','ก.ย','ต.ค','พ.ย','ธ.ค'];
  var full = ['มกราคม','กุมภาพันธ์','มีนาคม','เมษายน','พฤษภาคม','มิถุนายน',
              'กรกฎาคม','สิงหาคม','กันยายน','ตุลาคม','พฤศจิกายน','ธันวาคม'];
  var month = 0;
  for (var i = 0; i < 12; i++) {
    var a = abbr[i].replace(/\./g, '\\.?');
    if (new RegExp('(^|\\s)' + a + '\\.?(\\s|$|\\d)').test(n) || n.indexOf(full[i]) >= 0) { month = i + 1; break; }
  }
  var ym = n.match(/\b(25\d{2})\b/);
  var year = ym ? parseInt(ym[1], 10) : 0;
  if (!year) {                                   // ปีย่อ 2 หลัก เช่น "67" → 2567
    var y2 = n.match(/\b(\d{2})\b\s*$/) || n.match(/\s(\d{2})\b/);
    if (y2) year = 2500 + parseInt(y2[1], 10);
  }
  var label = (month ? abbr[month - 1] + '. ' : '') + (year || '');
  label = label.trim();
  // อ่านเดือนหรือปีไม่ได้ → ใช้ชื่อไฟล์เป็นป้ายกำกับ และส่ง 0 กลับไปให้ฝั่งเว็บรู้ว่าเชื่อถือไม่ได้
  if (!month || !year) label = n;
  return { month: month, year: year, label: label || n };
}

// ------------------------------------------------------------------- helpers
function toNum_(x) {
  if (x === null || x === undefined || x === '') return null;
  if (typeof x === 'number') return isNaN(x) ? null : x;
  var s = String(x).replace(/,/g, '').trim();
  if (s === '' || !/^-?\d+(\.\d+)?$/.test(s)) return null;
  var n = parseFloat(s);
  return isNaN(n) ? null : n;
}
function zeros_(n)   { var a = []; for (var i = 0; i < n; i++) a.push(0); return a; }
function sum_(a)     { var t = 0; for (var i = 0; i < a.length; i++) t += a[i] || 0; return t; }
function trim_(s)    { return String(s).trim(); }
function nonEmpty_(s){ return !!s; }
function allBlank_(row, col, n) {
  for (var i = 0; i < n; i++) if (toNum_(row[col + i]) !== null) return false;
  return true;
}
function effectiveUser_() {
  try { return Session.getEffectiveUser().getEmail() || '(ไม่ทราบ)'; } catch (e) { return '(ไม่ทราบ)'; }
}

/* CacheService จำกัดค่าละ 100KB — สรุปบางเดือนอาจเกิน จึงหั่นเป็นท่อน */
function putChunked_(cache, key, str) {
  var SIZE = 90000, parts = [];
  for (var i = 0; i < str.length; i += SIZE) parts.push(str.substring(i, i + SIZE));
  var map = {}; map[key + ':n'] = String(parts.length);
  for (var j = 0; j < parts.length; j++) map[key + ':' + j] = parts[j];
  cache.putAll(map, CACHE_SECS);
}
function getChunked_(cache, key) {
  var n = cache.get(key + ':n');
  if (!n) return null;
  n = parseInt(n, 10);
  var keys = []; for (var i = 0; i < n; i++) keys.push(key + ':' + i);
  var got = cache.getAll(keys), out = '';
  for (var j = 0; j < n; j++) {
    var part = got[key + ':' + j];
    if (part == null) return null;              // ท่อนหาย = แคชใช้ไม่ได้
    out += part;
  }
  return out;
}

/** ตอบกลับ — JSONP ถ้ามี callback (ข้ามปัญหา CORS ได้ทุกกรณี) */
function respond_(obj, callback) {
  var json = JSON.stringify(obj);
  if (callback) {
    var safe = String(callback).replace(/[^\w$]/g, '');
    return ContentService.createTextOutput(safe + '(' + json + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}

 * ★ รันฟังก์ชันนี้ 1 ครั้งจากปุ่ม Run ▶ ก่อน Deploy ★
 * Web App ที่ deploy แล้วจะเด้งขอสิทธิ์เองไม่ได้ ต้องอนุญาตจากหน้าแก้ไขก่อน
 * ดู Log เพื่อยืนยันว่ารันในนามบัญชีไหน และเห็นไฟล์กี่ไฟล์

  var scan = scanFolders_(DEFAULT_FOLDER_IDS);
  for (var i = 0; i < scan.folders.length; i++) {
    var f = scan.folders[i];
    Logger.log(f.error ? ('โฟลเดอร์ ' + f.id + ' → เปิดไม่ได้: ' + f.error)
                       : ('โฟลเดอร์ "' + f.name + '" → ' + f.matched + ' ไฟล์ (ข้าม ' + f.skipped + ')'));
  }
  Logger.log('พบไฟล์ข้อมูลรวม ' + scan.files.length + ' ไฟล์');
  if (scan.files.length) {
    // ทดลองอ่านแค่ไฟล์เดียวพอ — อ่านทั้งหมดจะเกินเวลาทำงาน
    var s1 = summarizeFile_(scan.files[scan.files.length - 1], true);
    Logger.log('ทดลองอ่าน "' + s1.name + '" → ' + s1.days.length + ' วัน, ' + s1.gates.length + ' ประตู');
  }
  Logger.log('อนุญาตสิทธิ์เรียบร้อย');
}
